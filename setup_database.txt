REVOKE CONNECT ON DATABASE postgres FROM PUBLIC;

-- Step 1: Create Database
CREATE DATABASE ekota_db;

-- Step 2: Connect to the newly created database
\c ekota_db

-- Required for uuid_generate_v4()
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TYPE "UserRole" AS ENUM ('ADMIN', 'RENTER', 'INVESTOR', 'PRODUCER');
CREATE TYPE "KycStatus" AS ENUM ('UNVERIFIED', 'PENDING', 'VERIFIED', 'REJECTED');
CREATE TYPE "CampaignStatus" AS ENUM ('DRAFT', 'FUNDING', 'FULLY_FUNDED', 'IN_PRODUCTION', 'DELIVERED');

-- Core User Accounts
CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(150) NOT NULL,
    phone_number VARCHAR(20),
    role "UserRole" NOT NULL,
    is_email_verified BOOLEAN NOT NULL DEFAULT FALSE,
    is_blocked BOOLEAN NOT NULL DEFAULT FALSE,
    blocked_reason TEXT,
    kyc_status "KycStatus" NOT NULL DEFAULT 'UNVERIFIED',
    didit_session_id VARCHAR(255),
    created_at TIMESTAMPTZ(6) NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ(6) NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- OTP store for registration and password reset
CREATE TABLE IF NOT EXISTS otp_verifications (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) NOT NULL,
    otp_code VARCHAR(10) NOT NULL,
    expires_at TIMESTAMPTZ(6) NOT NULL,
    is_used BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMPTZ(6) NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Listings created by producers
CREATE TABLE IF NOT EXISTS "Listing" (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    "assetName" TEXT NOT NULL,
    category TEXT NOT NULL,
    "fundingTarget" DOUBLE PRECISION NOT NULL,
    "rentalPrice" DOUBLE PRECISION NOT NULL,
    description TEXT NOT NULL,
    specifications TEXT,
    "imageUrls" TEXT[] NOT NULL DEFAULT ARRAY[]::TEXT[],
    "videoUrls" TEXT[] NOT NULL DEFAULT ARRAY[]::TEXT[],
    "productionTimeType" TEXT NOT NULL,
    "productionDays" INTEGER,
    status TEXT NOT NULL DEFAULT 'active',
    "campaignStatus" TEXT NOT NULL DEFAULT 'funding',
    upvotes INTEGER NOT NULL DEFAULT 0,
    downvotes INTEGER NOT NULL DEFAULT 0,
    "producerId" UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    "createdAt" TIMESTAMPTZ(6) NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updatedAt" TIMESTAMPTZ(6) NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Comments on listings, including optional producer replies
CREATE TABLE IF NOT EXISTS "Comment" (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    text TEXT NOT NULL,
    "listingId" UUID NOT NULL REFERENCES "Listing"(id) ON DELETE CASCADE,
    "authorId" UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    reply TEXT,
    "createdAt" TIMESTAMPTZ(6) NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Reviews left by investors/renters
CREATE TABLE IF NOT EXISTS "Review" (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    rating INTEGER NOT NULL,
    text TEXT NOT NULL,
    "listingId" UUID NOT NULL REFERENCES "Listing"(id) ON DELETE CASCADE,
    "authorId" UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    "createdAt" TIMESTAMPTZ(6) NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Helpful indexes for the current app flows
CREATE INDEX IF NOT EXISTS idx_users_email ON users(email);
CREATE INDEX IF NOT EXISTS idx_otp_verifications_email ON otp_verifications(email);
CREATE INDEX IF NOT EXISTS idx_listing_producer_id ON "Listing"("producerId");
CREATE INDEX IF NOT EXISTS idx_comment_listing_id ON "Comment"("listingId");
CREATE INDEX IF NOT EXISTS idx_review_listing_id ON "Review"("listingId");

-- Dev-only backend account
CREATE USER ekota_backend WITH PASSWORD 'ekota1234';

GRANT CONNECT ON DATABASE ekota_db TO ekota_backend;
GRANT USAGE ON SCHEMA public TO ekota_backend;
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public TO ekota_backend;
GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public TO ekota_backend;
ALTER USER ekota_backend WITH SUPERUSER;
